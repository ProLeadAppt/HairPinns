# Disallow query parameters that create duplicate content.
# Shopify variant/pagination/locale params generate near-infinite URL permutations
# that all canonicalise to the same product/collection page. Search Console
# (2026-05) flagged 182 URLs as "Alternative page with proper canonical tag" -
# canonicals were working but crawl budget was being wasted on the dupes.

Disallow: /*?*fbclid=*

Disallow: /*?*sort=*

Disallow: /*?*variant=*
